# SFL curl "https://xyloapi.qzz.io/api/downloader/sfile?url=https%3A%2F%2Fsfile.co%2Fxxxxx" # Bilibili curl "https://xyloapi.qzz.io/api/downloader/bilibili?url=https%3A%2F%2Fwww.bilibili.tv%2Fid%2Fvideo%2Fxxxxx"
// Website — Fetch API async function download(url, type='sfile') { const API = `https://xyloapi.qzz.io/api/downloader/${type}`; const res = await fetch(`${API}?url=${encodeURIComponent(url)}`); const json = await res.json(); if (!json.success) throw new Error('Gagal'); return json.data.links[0].url; }
// Node.js — npm install axios const axios = require('axios'); async function get(url, type='sfile') { const { data } = await axios.get(`https://xyloapi.qzz.io/api/downloader/${type}`, { params: { url } }); return data.data.links[0].url; }
# Python — pip install requests import requests def get(url, t='sfile'): r = requests.get(f'https://xyloapi.qzz.io/api/downloader/{t}', params={'url': url}) return r.json()['data']['links'][0]['url']
// PHP <?php function get($url, $t='sfile') { $ch = curl_init("https://xyloapi.qzz.io/api/downloader/$t?url=".urlencode($url)); curl_setopt($ch, CURLOPT_RETURNTRANSFER, true); $data = json_decode(curl_exec($ch), true); return $data['data']['links'][0]['url']; }
# Telegram Bot — pip install python-telegram-bot requests from telegram.ext import Application, MessageHandler, filters import requests async def handle(u, c): url = u.message.text.strip() t = 'sfile' if 'sfile.co' in url else 'bilibili' d = requests.get(f'https://xyloapi.qzz.io/api/downloader/{t}', params={'url': url}).json() await u.message.reply_text(d['data']['links'][0]['url']) app = Application.builder().token('TOKEN').build() app.add_handler(MessageHandler(filters.TEXT, handle)); app.run_polling()
// Discord — npm install discord.js axios const { Client, GatewayIntentBits } = require('discord.js'); const axios = require('axios'); const c = new Client({ intents: [GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages, GatewayIntentBits.MessageContent] }); c.on('messageCreate', async m => { const mt = m.content.match(/https?:\/\/\S+/); if (!mt) return; const t = mt[0].includes('sfile.co') ? 'sfile' : 'bilibili'; const { data } = await axios.get(`https://xyloapi.qzz.io/api/downloader/${t}`, { params: { url: mt[0] } }); m.reply(data.data.links[0].url); }); c.login('TOKEN');
